@@@QUERY: R0901_Forecas_SemAc
SELECT R09_Forecast_Semanal.Campaña, R09_Forecast_Semanal.Fundo, R09_Forecast_Semanal.[Mod], R09_Forecast_Semanal.Turno, R09_Forecast_Semanal.Lote, R09_Forecast_Semanal.Area, R09_Forecast_Semanal.FCosAnt, R09_Forecast_Semanal.FCos, R09_Forecast_Semanal.Sem, R09_Forecast_Semanal.FrtCos, R09_Forecast_Semanal.Rend, R09_Forecast_Semanal.Kg, R09_Forecast_Semanal.Dr, R09_Forecast_Semanal.Version, R09_Forecast_Semanal.FundPPTo
FROM R09_Forecast_Semanal
WHERE (
        ((R09_Forecast_Semanal.Campaña) IN("C2025", "C2026"))
        AND (
            (
                Int(
                    Right(Left([R09_Forecast_Semanal].[Version], 3), 2)
                )
            ) = [R09_Forecast_Semanal].[Sem]
        )
    );

@@@QUERY: R0902_Forecast_Sem_vs_Camp
SELECT *
FROM (SELECT "Proy."&R09_Forecast_Semanal.Version as Version, R09_Forecast_Semanal.Fundo, R09_Forecast_Semanal.[Mod], M_Time.Año, M_Time.Sem, Sum(R09_Forecast_Semanal.Kg) AS Kg, R09_Forecast_Semanal.Campaña
FROM R09_Forecast_Semanal INNER JOIN M_Time ON R09_Forecast_Semanal.FCos = M_Time.Fecha
GROUP BY R09_Forecast_Semanal.Version, R09_Forecast_Semanal.Fundo, R09_Forecast_Semanal.[Mod], M_Time.Año, M_Time.Sem, R09_Forecast_Semanal.Campaña

UNION ALL SELECT R08_Forecast_Campaña.Version, R08_Forecast_Campaña.FundoPPto, R08_Forecast_Campaña.Modulo, R08_Forecast_Campaña.Año, R08_Forecast_Campaña.Semana, R08_Forecast_Campaña.KG, R08_Forecast_Campaña.Campaña
FROM R08_Forecast_Campaña)  AS res;

@@@QUERY: R0903_Forecast_FrtsTotal
SELECT R09_Forecast_Semanal.Campaña, R09_Forecast_Semanal.Fundo, R09_Forecast_Semanal.[Mod], R09_Forecast_Semanal.Turno, R09_Forecast_Semanal.Lote, R09_Forecast_Semanal.Area, R09_Forecast_Semanal.FCosAnt, R09_Forecast_Semanal.FCos, R09_Forecast_Semanal.Sem, R09_Forecast_Semanal.FrtCos, R09_Forecast_Semanal.Rend, R09_Forecast_Semanal.Kg, R09_Forecast_Semanal.Dr, R09_Forecast_Semanal.Version, R09_Forecast_Semanal.FundPPTo, M_Lotes.NPlantas * R09_Forecast_Semanal.FrtCos AS frutosT
FROM R09_Forecast_Semanal INNER JOIN M_Lotes ON (R09_Forecast_Semanal.Lote = M_Lotes.Lote) AND (R09_Forecast_Semanal.[Mod] = M_Lotes.Modulo) AND (R09_Forecast_Semanal.Fundo = M_Lotes.Fundo_pptom5);

@@@QUERY: TPlantas
SELECT M_Lotes.FundoPPto, M_Lotes.Modulo, Sum(M_Lotes.NPlantas) AS TotalPlantas
FROM M_Lotes
GROUP BY M_Lotes.FundoPPto, M_Lotes.Modulo;
